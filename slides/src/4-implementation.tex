\begin{frame}{Implementation - Leader Behavior \& Kinematics}
    \begin{block}{Stochastic Leader Acceleration}
        \begin{equation*}
            a_L(t) =
            \begin{cases}
                \text{clip}\left(A \sin(\omega t + \phi), -8.0, 3.0\right) & \text{if } t < t_{brake} \\
                a_{max\_brake} & \text{if } t \ge t_{brake} \text{ and } v_L(t) > 0 \\
                0 & \text{otherwise}
            \end{cases}
        \end{equation*}
        \begin{itemize}
            \item $t_{brake} = \tau \sim \mathcal{U}(5, 20)$ with $p = 0.3$, else $\infty$.
            \item $A \sim \mathcal{U}(1.0, 5.0)\,\text{m/s}^2$, $\omega \sim \mathcal{U}(0.1, 0.4)\,\text{rad/s}$, $\phi \sim \mathcal{U}(0, 2\pi)$.
        \end{itemize}
    \end{block}
    \begin{block}{Kinematics Rules ($\Delta t = 0.1\,\text{s}$, discrete)}
        \begin{itemize}
            \item $v(t+1) = \max\left(0, v(t) + a(t) \cdot \Delta t\right)$
            \item $x(t+1) = x(t) + v(t) \cdot \Delta t + \frac{1}{2}a(t) \cdot \Delta t^2$
            \item Bounded acceleration: $-8.0 \,\text{m/s}^2 \le a(t) \le 3.0\,\text{m/s}^2$
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Implementation - State \& AI Module}
    \begin{block}{State Space $S_t \in \mathbb{R}^3$}
        \begin{itemize}
            \item $d(t) = x_L(t) - x_E(t)$ \ (Distance from LiDAR)
            \item $v_{rel}(t) = v_L(t) - v_E(t)$ \ (Relative velocity)
            \item $v_E(t)$ \ (Ego absolute velocity)
        \end{itemize}
    \end{block}
    \begin{block}{Initial Conditions ($t=0$)}
        \begin{itemize}
            \item $x_E(0) = 0\,\text{m}$, $d(0) \sim \mathcal{U}(5.0,20.0)\,\text{m}$
            \item $v_L(0), v_E(0) \sim \mathcal{U}(10.0, 20.0)\,\text{m/s}$, $a_E(-1) = 0.0\,\text{m/s}^2$
        \end{itemize}
    \end{block}
    \begin{block}{AI Module Structure}
        \begin{itemize}
            \item Multi-Layer Perceptron (MLP) mapping $S_t \to a_E(t)$.
            \item Hidden layers: $[16, 32, 32, 16]$ with ReLU activation.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Implementation - Safety Monitor \& Recovery}
    \begin{block}{Safety Monitor (Deterministic \& Assured)}
        Computes minimum distance required to avoid collision if the Leader brakes with maximum deceleration ($a_{max\_brake} = -8.0\,\text{m/s}^2$):
        \begin{equation*}
            d_{critical} = \max\left(0, v_E(t) \cdot \Delta t + \frac{v_E(t)^2 - v_L(t)^2}{2\cdot |a_{max\_brake}|}\right)
        \end{equation*}
        \begin{itemize}
            \item $v_E(t) \cdot \Delta t$ accounts for the system's 1-tick reaction delay.
            \item \textbf{Condition:} Safety Monitor takes control if $d(t) \le d_{critical}$.
        \end{itemize}
    \end{block}
    \begin{block}{Recovery Function}
        \begin{itemize}
            \item Fail-safe behavior: Apply maximum deceleration ($a_E = -8.0\,\text{m/s}^2$).
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Implementation - RL Environment \& Training}
    \begin{block}{Reward Function ($R_t = R_{distance} + R_{comfort} + R_{penalty}$)}
        \begin{itemize}
            \item $R_{distance} = -|d(t) - 10|$ (Keep target distance of $10\,\text{m}$).
            \item $R_{comfort} = -0.9 |a_E(t-1) - a_E(t)|$ (Penalize sudden changes).
            \item $R_{penalty} = -20$ if $d(t) \le d_{critical}$ (Penalize entering critical zone).
            \item $R_{terminal} = -1000$ on collision ($d(t) \le 0$) or lost leader ($d(t) > 50\,\text{m}$).
        \end{itemize}
    \end{block}
    \begin{block}{Tech Stack \& Hyperparameters}
        \begin{itemize}
            \item \textbf{Training:} SB3 PPO ($\gamma=0.99$, \texttt{batch\_size}=64, \texttt{n\_steps}=1024, \texttt{clip\_range}=0.2).
            \item \textbf{Integration:} C++ Simulator, \texttt{pybind11} for Python training, and \texttt{ONNX} runtime for embedded deployment.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Implementation - Simulation Results}
    \begin{block}{Evaluation of Scenarios}
        \begin{itemize}
            \item \textbf{Success:} The safety monitor successfully triggers recovery braking when the leader vehicle brakes suddenly, avoiding collision.
            \item \textbf{Collision / Leader Lost:} Boundaries are validated during training.
        \end{itemize}
    \end{block}
