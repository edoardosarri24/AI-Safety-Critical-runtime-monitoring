\subsection{Resiliency Real-Time and Certification}

\begin{frame}{Introduction}
    \begin{block}{AI in Safety-Critical Systems}
        Integrating AI models into systems requiring predictable and safe behavior:
        \begin{itemize}
            \item \textbf{Challenge:} Black-box nature of AI models.
            \item \textbf{Issue:} Lack of structured standards and methodologies.
        \end{itemize}
    \end{block}
    \begin{block}{Objective}
        Study and implement a Runtime Assurance (RTA) framework based on:
        \begin{itemize}
            \item \textit{``Artificial Intelligence in safety-critical systems''} (Perez, 2024).
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Introduction - Terminology}
    \begin{block}{Key Concepts}
        \begin{itemize}
            \item \textbf{AI and Machine Learning:} Algorithmic techniques allowing models to learn patterns from data.
            \item \textbf{AI Safety:} Methodologies to deploy AI in safety-critical systems in compliance with safety standards, avoiding/mitigating catastrophic harm.
            \item \textbf{Functional Safety (FuSa):} ``Part of the overall safety of a system that assures the freedom from unacceptable risk'' (Perez, 2024).
        \end{itemize}
    \end{block}
\end{frame}
